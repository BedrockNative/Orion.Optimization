cmake_minimum_required(VERSION 3.28)
project(OrionOptimization VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use clang-cl targeting Windows x64 (cmake/clang-cl-x64.cmake on Linux).")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    message(FATAL_ERROR "This scaffold requires clang-cl.")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(BUILD_SHARED_LIBS OFF)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _ITERATOR_DEBUG_LEVEL=0)
add_compile_options("$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Release>:/Ob3>" "$<$<CONFIG:Release>:/clang:-O3>")
set(LIBHAT_TESTING OFF CACHE BOOL "" FORCE)
set(LIBHAT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBHAT_INSTALL_TARGET OFF CACHE BOOL "" FORCE)
add_subdirectory(external/libhat EXCLUDE_FROM_ALL)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(external/zydis EXCLUDE_FROM_ALL)
# Point find_package at the Zydis target already built from our submodule.
set(SAFETYHOOK_FETCH_ZYDIS OFF CACHE BOOL "" FORCE)
set(Zydis_DIR "${PROJECT_SOURCE_DIR}/cmake/packages/Zydis" CACHE PATH "" FORCE)
add_subdirectory(external/SafetyHook EXCLUDE_FROM_ALL)
set(GSL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(external/GSL EXCLUDE_FROM_ALL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(cmake/RuntimeImporter.cmake)
file(GLOB_RECURSE GAME_HEADERS CONFIGURE_DEPENDS include/mc/*.h)
set(IMPORT_DIR "${CMAKE_CURRENT_BINARY_DIR}/imports")
set(IMPORT_LIB "${IMPORT_DIR}/win-client/Minecraft.Windows.lib")
add_custom_command(OUTPUT "${IMPORT_LIB}" "${IMPORT_DIR}/win-client/symbols/orion.symbols.json"
    COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/generate_imports.py"
        --root "${PROJECT_SOURCE_DIR}" --out "${IMPORT_DIR}" --sdk "${XWIN_ROOT}" --librarian "${CMAKE_AR}"
    DEPENDS ${GAME_HEADERS} tools/generate_imports.py VERBATIM)
add_custom_target(game_imports DEPENDS "${IMPORT_LIB}")
file(GLOB_RECURSE IMPORTER_SOURCES CONFIGURE_DEPENDS vendor/amethyst/amethyst/runtime/importing/*.cpp)
add_library(OrionOptimization SHARED src/mod.cpp src/memory.cpp src/optimization.cpp src/parsed-cache.cpp src/game-helpers.cpp ${IMPORTER_SOURCES})
add_dependencies(OrionOptimization game_imports)
target_include_directories(OrionOptimization PRIVATE include vendor/amethyst)
target_compile_options(OrionOptimization PRIVATE /W4 /EHsc /Zi /permissive-)
target_link_libraries(OrionOptimization PRIVATE libhat::libhat safetyhook::safetyhook Microsoft.GSL::GSL "${IMPORT_LIB}")
target_link_options(OrionOptimization PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
set_target_properties(OrionOptimization PROPERTIES OUTPUT_NAME "Orion.Optimization" PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    LINK_DEPENDS "${IMPORT_DIR}/win-client/symbols/orion.symbols.json")
add_custom_command(TARGET OrionOptimization POST_BUILD
    COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/postprocess.py"
        --dotnet "${DOTNET_EXECUTABLE}" --tool "${ORION_RUNTIME_IMPORTER_DIR}/Amethyst.ModuleTweaker.dll"
        --module "$<TARGET_FILE:OrionOptimization>" --input "${IMPORT_DIR}" --output "${CMAKE_BINARY_DIR}/postprocess"
    VERBATIM)

option(ORION_BUILD_TESTS "Build the isolated Windows import-resolution fixture" OFF)
if(ORION_BUILD_TESTS)
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/fixture.hpp"
        COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/generate_fixture.py"
            "${IMPORT_DIR}/win-client/symbols/orion.symbols.json" "${CMAKE_BINARY_DIR}/fixture.hpp"
        DEPENDS "${IMPORT_LIB}" tools/generate_fixture.py VERBATIM)
    add_executable(importer_smoke tests/importer_smoke.cpp "${CMAKE_BINARY_DIR}/fixture.hpp")
    target_link_options(importer_smoke PRIVATE /SECTION:.orion,ERW)
    target_include_directories(importer_smoke PRIVATE "${CMAKE_BINARY_DIR}")
    set_target_properties(importer_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(ORION_BUILD_TESTS)
    add_executable(query_cache_tests tests/query_cache.cpp)
    target_include_directories(query_cache_tests PRIVATE include)
    target_compile_options(query_cache_tests PRIVATE /EHsc)
    add_executable(hook_abi_tests tests/hook_abi.cpp)
    target_link_libraries(hook_abi_tests PRIVATE safetyhook::safetyhook Microsoft.GSL::GSL)
    target_compile_options(hook_abi_tests PRIVATE /EHsc)
    set_target_properties(query_cache_tests hook_abi_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(ORION_BUILD_TESTS)
    add_executable(pack_changes_tests tests/pack_changes.cpp)
    target_include_directories(pack_changes_tests PRIVATE include)
    target_compile_options(pack_changes_tests PRIVATE /EHsc)
    set_target_properties(pack_changes_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
